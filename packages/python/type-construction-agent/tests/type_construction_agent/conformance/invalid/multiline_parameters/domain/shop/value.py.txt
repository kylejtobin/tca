from pydantic import BaseModel, ConfigDict, RootModel

    """A thing."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True, validate_default=True, revalidate_instances="never", from_attributes=True)
    @property
    def total(
        self,
        rate: int,
    ) -> int:
        return rate
