from pydantic import BaseModel, ConfigDict, RootModel

from domain.shop.value import Lines


class Order(BaseModel):
    """A request to buy some lines."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True, validate_default=True, revalidate_instances="never", from_attributes=True)
    lines: Lines

    @property
    def charge(self) -> "Charge":
        return Charge(order=self)


class Charge(BaseModel):
    """The intended charging of an order."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True, validate_default=True, revalidate_instances="never", from_attributes=True)
    order: Order


class Paid(BaseModel):
    """An order that was paid."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True, validate_default=True, revalidate_instances="never", from_attributes=True)
    order: Order


class Declined(BaseModel):
    """An order that was declined."""

    model_config = ConfigDict(frozen=True, extra="forbid", strict=True, validate_default=True, revalidate_instances="never", from_attributes=True)
    lines: Lines


class Outcome(RootModel[Paid | Declined]):
    """What became of an order."""

    model_config = ConfigDict(frozen=True, strict=True, validate_default=True, revalidate_instances="never")
