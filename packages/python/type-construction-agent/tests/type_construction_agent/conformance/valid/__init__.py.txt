import asyncio

from pydantic_ai import CancellationToken
from pydantic_settings import EnvSettingsSource

from config import ShopConfig
from domain.shop.order import Order
from integration.shop.interpreter import ChargeInterpreter
from integration.shop.model import ChargeReply

__all__ = ("charge", "configure")


def configure() -> ShopConfig:
    return ShopConfig.model_validate_strings(EnvSettingsSource(ShopConfig)())


async def charge(config: ShopConfig, order: Order, *, cancellation_token: CancellationToken | None = None) -> ChargeReply:
    return await ChargeInterpreter(action=order.charge, provider=config.shop).interpret(cancellation_token=cancellation_token)


def charge_sync(config: ShopConfig, order: Order) -> ChargeReply:
    return asyncio.run(charge(config, order))
